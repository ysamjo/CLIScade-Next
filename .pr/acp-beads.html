Beads → ACP → gespeicherter Handoff

Neu ist die fachliche Verbindung zwischen Beads und OpenHands. Vorher konnte Canvas ACP-Conversations starten; Beads-Aufgaben, Attempt-Zuordnung und portable Checkpoints fehlten.

Ausführungsgrenze: Vor der Prompt-Übermittlung wird der Attempt dauerhaft markiert. Unklarer Versand führt zu Prüfung statt automatischem Replay.

+"""Beads bleibt die einzige lesbare und schreibbare Aufgabenquelle."""
+
+import json
+import os
+import re
+import subprocess
+from pathlib import Path
+
+from filelock import FileLock
+
+
+class Beads:
+    @staticmethod
+    def validate_id(task_id):
+        if not isinstance(task_id, str) or not re.fullmatch(
+            r"[a-zA-Z][a-zA-Z0-9_.-]{1,150}", task_id
+        ):
+            raise ValueError("Ungültige Beads-ID")
+
+    def __init__(self, project: Path):
+        self.project = project.resolve()
+        directory = self.project / ".beads"
+        if not directory.is_dir() or directory.is_symlink():
+            raise ValueError("Beads ist im Projekt nicht eingerichtet")
+        self.directory = directory
+
+    def call(self, *args: str, readonly=False, json_output=True):
+        command = ["bd", *args, "--sandbox"]
+        if readonly:
+            command.append("--readonly")
+        if json_output:
+            command.append("--json")
+        env = {**os.environ, "BEADS_DIR": str(self.directory)}
+        with FileLock(str(self.directory / "cliscade.lock"), timeout=30):
+            result = subprocess.run(
+                command,
+                cwd=self.project,
+                env=env,
+                capture_output=True,
+                text=True,
+                timeout=30,
+                check=False,
+            )
+        if result.returncode:
+            raise RuntimeError("Beads-Zugriff fehlgeschlagen; bd doctor prüfen")
+        return json.loads(result.stdout) if json_output else result.stdout
+
+    def tasks(self):
+        return self.call("list", "--all", "--limit", "0", readonly=True)
+
+    def show(self, task_id):
+        self.validate_id(task_id)
+        result = self.call("show", task_id, readonly=True)
+        task = result[0] if isinstance(result, list) and result else result
+        if not isinstance(task, dict) or task.get("id") != task_id:
+            raise ValueError("Beads-Aufgabe nicht gefunden")
+        return task
+
+    def create(self, title, description="", acceptance="", priority=2, parent=None):
+        args = [
+            "create",
+            "--title",
+            title,
+            "--description",
+            description,
+            "--acceptance",
+            acceptance,
+            "--priority",
+            str(priority),
+        ]
+        if parent:
+            self.validate_id(parent)
+            args.extend(["--parent", parent])
+        return self.call(*args)
+
+    def update(self, task_id, **fields):
+        self.validate_id(task_id)
+        allowed = {
+            "title",
+            "description",
+            "acceptance",
+            "priority",
+            "assignee",
+            "status",
+        }
+        if not fields or fields.keys() - allowed:
+            raise ValueError("Ungültige Aufgabenänderung")
+        args = ["update", task_id]
+        for key, value in fields.items():
+            args.extend(["--" + key, str(value)])
+        return self.call(*args)
+
+    def dependency(self, task_id, depends_on):
+        self.validate_id(task_id)
+        self.validate_id(depends_on)
+        return self.call("dep", "add", task_id, depends_on)
+
+    def assign(self, task_id, agent_id, previous=None):
+        self.validate_id(task_id)
+        args = ["update", task_id, "--assignee", agent_id, "--status", "in_progress"]
+        if previous:
+            args.extend(
+                ["--if-assignee", previous, "--if-status", "in_progress"]
+            )
+        return self.call(*args)
+
+    def context(self):
+        return {
+            "workflow": self.call(
+                "prime", "--full", "--no-memories", readonly=True, json_output=False
+            ),
+            "memories": self.call("memories", readonly=True),
+        }

+"""ACP-Integration ausschließlich über den offiziellen OpenHands-Client."""
+
+from dataclasses import dataclass
+from pathlib import Path
+from uuid import UUID
+
+from openhands.sdk import RemoteConversation
+from openhands.sdk.agent import ACPAgent
+from openhands.sdk.settings.acp_providers import ACP_PROVIDERS
+from openhands.sdk.workspace import RemoteWorkspace
+
+
+@dataclass(frozen=True)
+class Agent:
+    id: str
+    command: tuple[str, ...]
+    model: str | None = None
+
+
+def registry():
+    agents = {
+        key: Agent(key, tuple(ACP_PROVIDERS[key].default_command))
+        for key in ("codex", "claude-code", "opencode")
+    }
+    agents["kiro"] = Agent("kiro", ("kiro-cli", "acp"))
+    return agents
+
+
+class OpenHands:
+    def __init__(self, settings):
+        self.settings = settings
+
+    def attach(self, agent: Agent, workspace: Path, session_id=None):
+        remote = RemoteWorkspace(
+            host=self.settings.agent_server,
+            working_dir=str(workspace),
+            api_key=self.settings.api_key,
+        )
+        try:
+            conversation = (
+                RemoteConversation.attach(
+                    workspace=remote,
+                    conversation_id=UUID(session_id),
+                    visualizer=None,
+                )
+                if session_id
+                else RemoteConversation(
+                    agent=ACPAgent(
+                        acp_command=list(agent.command), acp_model=agent.model
+                    ),
+                    workspace=remote,
+                    visualizer=None,
+                    delete_on_close=False,
+                    tags={"cliscade": "v1", "agent": agent.id},
+                )
+            )
+        except BaseException:
+            remote.client.close()
+            raise
+        return Session(conversation, remote)
+
+
+class Session:
+    def __init__(self, conversation, workspace):
+        self.conversation = conversation
+        self.workspace = workspace
+        self.id = str(conversation.id)
+
+    def submit(self, prompt):
+        self.conversation.send_message(prompt)
+        self.conversation.run(blocking=False)
+
+    def state(self):
+        state = self.conversation.state
+        return {
+            "status": str(state.execution_status.value),
+            "events": [event.model_dump(mode="json") for event in state.events],
+        }
+
+    def cancel(self):
+        self.conversation.interrupt()
+
+    def close(self):
+        self.conversation.close()
+        self.workspace.client.close()

+"""Dauerhafte Ausführungszuordnung ohne zweite Aufgaben-Datenbank."""
+
+import json
+import sqlite3
+from contextlib import contextmanager
+from pathlib import Path
+from uuid import uuid4
+
+
+class Ledger:
+    def __init__(self, directory: Path):
+        directory.mkdir(parents=True, exist_ok=True)
+        self.path = directory / "attempts.sqlite"
+        with self.connect() as db:
+            db.executescript("""
+                PRAGMA journal_mode=WAL;
+                CREATE TABLE IF NOT EXISTS attempts (
+                    id TEXT PRIMARY KEY, project TEXT NOT NULL,
+                    task_id TEXT NOT NULL, agent TEXT NOT NULL,
+                    workspace TEXT NOT NULL, status TEXT NOT NULL,
+                    session_id TEXT, prompt_sent INTEGER NOT NULL DEFAULT 0,
+                    source_id TEXT, checkpoint TEXT, error TEXT, terminal_seen REAL,
+                    created_at TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ'))
+                );
+                CREATE UNIQUE INDEX IF NOT EXISTS active_workspace
+                ON attempts(workspace)
+                WHERE status IN ('preparing','submitted','running','cancelling');
+            """)
+
+    @contextmanager
+    def connect(self):
+        db = sqlite3.connect(self.path, timeout=30)
+        db.row_factory = sqlite3.Row
+        try:
+            with db:
+                yield db
+        finally:
+            db.close()
+
+    def create(self, project, task_id, agent, workspace, source_id=None):
+        attempt_id = str(uuid4())
+        with self.connect() as db:
+            db.execute(
+                "INSERT INTO attempts (id,project,task_id,agent,workspace,status,"
+                "source_id) VALUES (?,?,?,?,?,'preparing',?)",
+                (attempt_id, project, task_id, agent, str(workspace), source_id),
+            )
+        return self.get(attempt_id)
+
+    def get(self, attempt_id):
+        with self.connect() as db:
+            row = db.execute(
+                "SELECT * FROM attempts WHERE id=?", (attempt_id,)
+            ).fetchone()
+        if row is None:
+            raise ValueError("Ausführung nicht gefunden")
+        item = dict(row)
+        if item["checkpoint"]:
+            item["checkpoint"] = json.loads(item["checkpoint"])
+        return item
+
+    def list(self, project, task_id=None):
+        sql = "SELECT id FROM attempts WHERE project=?"
+        args = [project]
+        if task_id:
+            sql += " AND task_id=?"
+            args.append(task_id)
+        with self.connect() as db:
+            rows = db.execute(sql + " ORDER BY created_at,id", args).fetchall()
+        return [self.get(row["id"]) for row in rows]
+
+    def update(self, attempt_id, **fields):
+        allowed = {
+            "status",
+            "session_id",
+            "prompt_sent",
+            "checkpoint",
+            "error",
+            "terminal_seen",
+        }
+        if not fields or fields.keys() - allowed:
+            raise ValueError("Ungültige Ausführungsänderung")
+        if "checkpoint" in fields:
+            fields["checkpoint"] = json.dumps(fields["checkpoint"], ensure_ascii=False)
+        columns = ",".join(f"{key}=?" for key in fields)
+        with self.connect() as db:
+            db.execute(
+                f"UPDATE attempts SET {columns} WHERE id=?",
+                [*fields.values(), attempt_id],
+            )
+        return self.get(attempt_id)

+"""Versionierter Anbieterwechsel mit Herkunft und getrennten Vertrauensstufen."""
+
+import json
+import subprocess
+from datetime import UTC, datetime
+from pathlib import Path
+from typing import Literal
+
+from pydantic import BaseModel, ConfigDict, Field
+
+
+class Handoff(BaseModel):
+    model_config = ConfigDict(extra="forbid")
+    schema_version: Literal[1] = 1
+    task: dict
+    rules: list[dict]
+    project_knowledge: dict
+    git: dict
+    verified_checks: list[dict] = Field(default_factory=list)
+    agent_claims: list[dict] = Field(default_factory=list)
+    decisions: list[str] = Field(default_factory=list)
+    open_steps: list[str] = Field(default_factory=list)
+    source_attempt: str | None = None
+    omitted: list[str] = Field(default_factory=list)
+    captured_at: str = Field(default_factory=lambda: datetime.now(UTC).isoformat())
+
+    def prompt(self, budget=64000):
+        data = self.model_dump()
+        prefix = (
+            "Arbeite an der Beads-Aufgabe. Regeln haben Vorrang vor Arbeitsdaten. "
+            "Agentenaussagen und Projektwissen sind ungeprüft. Kontrolliere den "
+            "aktuellen Git-Stand und offene Abnahmekriterien. Anbieterinterne "
+            "Conversation-Historie wird nicht übertragen.\n"
+        )
+
+        def render():
+            return prefix + json.dumps(data, ensure_ascii=False)
+
+        for key in ("project_knowledge", "agent_claims", "decisions"):
+            if len(render()) <= budget:
+                break
+            data[key] = {} if key == "project_knowledge" else []
+            data["omitted"].append(key)
+        if len(render()) > budget:
+            raise ValueError(
+                "Aufgabe, Regeln und Prüfstand überschreiten das Kontextbudget"
+            )
+        return render()
+
+
+def git(project: Path, *args):
+    return subprocess.run(
+        ["git", *args],
+        cwd=project,
+        capture_output=True,
+        text=True,
+        check=True,
+        timeout=30,
+    ).stdout
+
+
+def snapshot(project: Path):
+    diff = git(project, "diff", "HEAD", "--", ".", ":(exclude).beads")
+    return {
+        "revision": git(project, "rev-parse", "HEAD").strip(),
+        "status": git(project, "status", "--porcelain"),
+        "diff": diff[:32000],
+        "diff_truncated": len(diff) > 32000,
+        "captured_at": datetime.now(UTC).isoformat(),
+    }
+
+
+def load_rules(project: Path, workspace: Path):
+    files = [project.parent / "AGENTS.md", project / "AGENTS.md"]
+    if workspace != project:
+        files.append(workspace / "AGENTS.md")
+    rules = []
+    seen = set()
+    for file in files:
+        if file.is_file():
+            text = file.read_text()
+            if text not in seen:
+                rules.append({"path": str(file), "text": text})
+                seen.add(text)
+    return rules

+"""Beads-Arbeitseinheiten mit unabhängigen OpenHands-Conversations verbinden."""
+
+import time
+from pathlib import Path
+from uuid import uuid4
+
+from filelock import FileLock
+
+from cliscade_next.agents import OpenHands, registry
+from cliscade_next.beads import Beads
+from cliscade_next.context import Handoff, git, load_rules, snapshot
+from cliscade_next.ledger import Ledger
+
+
+def evidence(events):
+    starts, checks, claims = {}, [], []
+    for event in events:
+        action = event.get("action") or {}
+        if event.get("kind") == "ActionEvent" and action.get("kind") == "FinishAction":
+            claims.append(
+                {
+                    "verified": False,
+                    "event_id": event.get("id"),
+                    "message": action.get("message", ""),
+                }
+            )
+        if event.get("kind") == "ACPToolCallEvent":
+            tool_id = event.get("tool_call_id")
+            raw = event.get("raw_input")
+            if event.get("status") in ("pending", "in_progress") and isinstance(
+                raw, dict
+            ):
+                command = raw.get("command")
+                if isinstance(command, str):
+                    starts[tool_id] = command
+            output = event.get("raw_output")
+            if tool_id in starts and isinstance(output, dict):
+                code = output.get("exit_code", output.get("exitCode"))
+                if type(code) is int and event.get("status") in ("completed", "failed"):
+                    checks.append(
+                        {
+                            "command": starts.pop(tool_id),
+                            "exit_code": code,
+                            "event_id": event.get("id"),
+                            "verification": "tool_receipt",
+                            "independently_reproduced": False,
+                            "timestamp": event.get("timestamp"),
+                        }
+                    )
+        if event.get("kind") == "MessageEvent" and event.get("source") == "agent":
+            claims.append(
+                {
+                    "verified": False,
+                    "event_id": event.get("id"),
+                    "message": event.get("llm_message", {}),
+                }
+            )
+    return checks, claims
+
+
+class Orchestrator:
+    def __init__(
+        self, settings, runner=None, beads_factory=Beads, agents=None, settle_seconds=30
+    ):
+        self.settings = settings
+        self.ledger = Ledger(settings.state)
+        self.runner = runner or OpenHands(settings)
+        self.beads_factory = beads_factory
+        self.agents = agents if agents is not None else registry()
+        self.settle_seconds = settle_seconds
+
+    def checkpoint(self, attempt, events=()):
+        project = self.settings.project(attempt["project"])
+        beads = self.beads_factory(project)
+        workspace = Path(attempt["workspace"])
+        checks, claims = evidence(events)
+        old = attempt.get("checkpoint") or {}
+        return Handoff(
+            task=beads.show(attempt["task_id"]),
+            rules=load_rules(project, workspace),
+            project_knowledge=beads.context(),
+            git=snapshot(workspace),
+            verified_checks=checks or old.get("verified_checks", []),
+            agent_claims=claims or old.get("agent_claims", []),
+            decisions=old.get("decisions", []),
+            open_steps=old.get("open_steps")
+            or ["Abnahmekriterien am Git-Stand prüfen"],
+            source_attempt=attempt.get("source_id"),
+        ).model_dump()
+
+    def start(self, project_name, task_id, agent_id, source_id=None):
+        if agent_id not in self.agents:
+            raise ValueError("ACP-Agent ist nicht freigegeben")
+        project = self.settings.project(project_name)
+        self.settings.state.mkdir(parents=True, exist_ok=True)
+        with FileLock(str(self.settings.state / "dispatch.lock"), timeout=30):
+            source = self.refresh(source_id) if source_id else None
+            if source:
+                if (source["project"], source["task_id"]) != (project_name, task_id):
+                    raise ValueError("Übergabe gehört zu einer anderen Aufgabe")
+                if source["status"] in ("uncertain", "preparing", "needs_review"):
+                    raise ValueError(
+                        "Unsicherer Stand: zuerst Session und Dateien prüfen"
+                    )
+                if source["status"] not in ("completed", "failed", "cancelled"):
+                    raise ValueError("Ausführung vor der Übergabe anhalten")
+                workspace = Path(source["workspace"])
+            else:
+                workspace = (
+                    self.settings.projects / ".cliscade-worktrees" / str(uuid4())
+                )
+                workspace.parent.mkdir(parents=True, exist_ok=True)
+                git(project, "worktree", "add", "--detach", str(workspace), "HEAD")
+            attempt = self.ledger.create(
+                project_name, task_id, agent_id, workspace, source_id
+            )
+            session = None
+            try:
+                checkpoint = self.checkpoint(attempt)
+                if source:
+                    checkpoint.update(
+                        {
+                            key: source["checkpoint"][key]
+                            for key in (
+                                "verified_checks",
+                                "agent_claims",
+                                "decisions",
+                                "open_steps",
+                            )
+                        }
+                    )
+                    checkpoint["source_attempt"] = source_id
+                prompt = Handoff.model_validate(checkpoint).prompt()
+                self.ledger.update(attempt["id"], checkpoint=checkpoint)
+                self.beads_factory(project).assign(
+                    task_id, agent_id, previous=source["agent"] if source else None
+                )
+                session = self.runner.attach(self.agents[agent_id], workspace)
+                self.ledger.update(
+                    attempt["id"],
+                    session_id=session.id,
+                    prompt_sent=1,
+                    status="submitted",
+                )
+                session.submit(prompt)
+                return self.ledger.update(attempt["id"], status="running")
+            except Exception:
+                saved = self.ledger.get(attempt["id"])
+                return self.ledger.update(
+                    attempt["id"],
+                    status="uncertain"
+                    if saved["prompt_sent"]
+                    else "preparation_failed",
+                    error="Ausführung fehlgeschlagen; Session und Arbeitsstand prüfen",
+                )
+            finally:
+                if session:
+                    session.close()
+
+    def refresh(self, attempt_id):
+        attempt = self.ledger.get(attempt_id)
+        if attempt["status"] in (
+            "completed",
+            "failed",
+            "cancelled",
+            "preparation_failed",
+        ):
+            return attempt
+        if not attempt["session_id"]:
+            return self.ledger.update(
+                attempt_id,
+                status="needs_review",
+                error="Vorbereitung unterbrochen; kein automatischer Neustart",
+            )
+        session = self.runner.attach(
+            self.agents[attempt["agent"]],
+            Path(attempt["workspace"]),
+            attempt["session_id"],
+        )
+        try:
+            state = session.state()
+            status = state["status"]
+            if status in ("finished", "error", "stuck", "paused"):
+                seen = attempt["terminal_seen"]
+                if status == "finished" and (
+                    seen is None or time.time() - seen < self.settle_seconds
+                ):
+                    if seen is None:
+                        self.ledger.update(attempt_id, terminal_seen=time.time())
+                    if self.settle_seconds:
+                        return self.ledger.get(attempt_id)
+                terminal = {
+                    "finished": "completed",
+                    "error": "failed",
+                    "stuck": "failed",
+                    "paused": "cancelled"
+                    if attempt["status"] == "cancelling"
+                    else "needs_review",
+                }
+                return self.ledger.update(
+                    attempt_id,
+                    status=terminal[status],
+                    checkpoint=self.checkpoint(attempt, state["events"]),
+                )
+            if status == "running":
+                return self.ledger.update(
+                    attempt_id, terminal_seen=None, status="running"
+                )
+            return attempt
+        finally:
+            session.close()
+
+    def cancel(self, attempt_id):
+        attempt = self.ledger.get(attempt_id)
+        if not attempt["session_id"] or attempt["status"] in (
+            "completed",
+            "failed",
+            "cancelled",
+        ):
+            raise ValueError("Keine aktive Session vorhanden")
+        self.ledger.update(attempt_id, status="cancelling")
+        session = self.runner.attach(
+            self.agents[attempt["agent"]],
+            Path(attempt["workspace"]),
+            attempt["session_id"],
+        )
+        try:
+            session.cancel()
+        finally:
+            session.close()
